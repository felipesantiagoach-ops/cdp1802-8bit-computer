; that's (9 X 256) +(15 X 16) + 8 = 2552 decimal

:LIST

10 PRINT "STARTING HOLLYWOOD COMPUTER DEMO"
20 LET N=10
30 PRINT "N IS "; N
40 REM CALL $38 TO OUTPUT N ON PORT 4, (LEDs)
50 LET X=USR(38,4,N)
60 LET N=N-1
70 IF N>0 GOTO 30
90 END

:X=USR(2552)

NEW? Y
:LIST
